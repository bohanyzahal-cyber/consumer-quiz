# יחידה 5. כל עובדה כאן מופיעה בסיכום המצגת ובבנק השאלות src/bank5.js.
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק חֲמִישִׁי: מְעֹרָבוּת הַצַרְכָנִים. מַצֶגֶת חָמֵשׁ. בִּרְשִׁימַת הַנוֹשְׂאִים לַבְּחִינָה יֵשׁ שְׁנֵי דְבָרִים: הַגוֹרְמִים שֶׁמַשְׁפִּיעִים עַל הַמְעֹרָבוּת, וּדְפוּסֵי הַקְנִיָה לְפִי מוֹדֵל עֲשָׂהאֵל. נַתְחִיל מֵהַהַגְדָרָה.

א: מְעֹרָבוּת הִיא רָמַת הַנְגִיעָה הָאִישִׁית שֶׁאָדָם מְיַחֵס לְמוּצָר, אוֹ לְהִתְנַהֲגוּת מְסֻיֶמֶת. בְּמִלִים פְּשׁוּטוֹת: כַּמָה הַמוּצָר חָשׁוּב וְרֵלֵוַנְטִי לַצַרְכָן.

ש: וְזֶה מִשְׁתַנֶה שֶׁל כֵּן וְלֹא?

א: לֹא. זֶה מִשְׁתַנֶה רָצִיף, שֶׁנָע מִמְעֹרָבוּת נְמוּכָה עַד מְעֹרָבוּת גְבוֹהָה. וּלְפִי רָמַת הַמְעֹרָבוּת נִקְבַּע כַּמָה זְמַן וּמַאֲמָץ הָאָדָם מוּכָן לְהַשְׁקִיעַ בְּכֹל שָׁלָב.

ש: וְאֵיךְ מוֹדְדִים מְעֹרָבוּת?

א: בְּסֻלַם הַמְעֹרָבוּת הָאִישִׁית שֶׁל זַיְקוֹבְסְקִי. זֶה שְׁאֵלוֹן עִם הַרְבֵּה פְּרִיטִים. בַּמַחֲצִית מֵהַפְּרִיטִים יֵשׁ כּוֹכָבִית: שָׁם הוֹפְכִים אֶת הַתְשׁוּבָה. וְאַחַר כָּךְ מְחַשְׁבִים מְמֻצָע.

ש: וּמָה הַמָקוֹם שֶׁל הַמְעֹרָבוּת בְּתַהֲלִיךְ הַהַחְלָטָה?

א: הִיא לֹא עוֹד שָׁלָב בַּתַהֲלִיךְ. הִיא מִשְׁתַנֶה מִתְעָרֵב, שֶׁמַשְׁפִּיעַ עַל כֹּל הַשְׁלַבִּים שֶׁבָּאִים אַחֲרֵי הַהַכָּרָה בַּצֹרֶךְ.

ש: אָז נַשְׁוֶה. מְעֹרָבוּת גְבוֹהָה.

א: תַהֲלִיךְ הַהַחְלָטָה מַקִיף, וְנַעֲשָׂה מֵרֹאשׁ. עִבּוּד הַמֵידָע מַעֲמִיק, בַּזִכָּרוֹן הַמְפוֹרָשׁ. הָעֶמְדָה שֶׁמִתְגַבֶּשֶׁת לִפְנֵי הַקְנִיָה יַצִיבָה מְאוֹד, וְיֵשׁ מִתְאָם גָבוֹהַ בֵּינָה לְבֵין הַהִתְנַהֲגוּת בִּנְקֻדַת הָרְכִישָׁה.

ש: וּמָה זֶה אוֹמֵר עַל מָה שֶׁקוֹרֶה בַּחֲנוּת?

א: שֶׁהַצַרְכָן פָּחוֹת מֻשְׁפָּע מִגֵרוּיִים בִּנְקֻדַת הָרְכִישָׁה. נְגִישׁוּת עַל הַמַדָף, מִבְצָעִים שֶׁל הַמִתְחָרִים, שִׁנוּיֵי מְחִיר. הוּא כְּבָר הֶחֱלִיט, עַל בְּסִיס שִׁקוּלִים מַהוּתִיִים.

ש: וְאַחֲרֵי הַקְנִיָה?

א: כְּכֹל שֶׁהַמְעֹרָבוּת גְבוֹהָה יוֹתֵר, הַהַעֲרָכָה שֶׁאַחֲרֵי הָרְכִישָׁה מַשְׁמָעוּתִית יוֹתֵר. יֵשׁ יוֹתֵר לְבָטִים, וּפּוֹטֶנְצְיָאל גָבוֹהַ יוֹתֵר לְדִיסוֹנַנְס.

ש: וּמְעֹרָבוּת נְמוּכָה.

א: גַם שָׁם יֵשׁ תַהֲלִיךְ שֶׁל קַבָּלַת הַחְלָטָה. אֲבָל הוּא קָצָר, וְהוּא יָכוֹל לְהִתְבַּצֵעַ בִּנְקֻדַת הָרְכִישָׁה עַצְמָה. עִבּוּד הַמֵידָע מָהִיר וְאוֹטוֹמָטִי, בַּזִכָּרוֹן הַמְרֻמָז.

ש: עַכְשָׁו הַנוֹשֵׂא הָרִאשׁוֹן לַבְּחִינָה: מָה קוֹבֵעַ אֶת רָמַת הַמְעֹרָבוּת. שְׁלוֹשָׁה גוֹרְמִים עִקָרִיִים. הָרִאשׁוֹן.

א: הַמוּצָר. כַּמָה תוֹעָלוֹת אוֹ סִכּוּנִים כְּרוּכִים בּוֹ, מִבְּחִינַת הַצַרְכָן. וְיֵשׁ חֲמִשָׁה סוּגִים שֶׁל סִכּוּנִים: כַּלְכָּלִי, חֶבְרָתִי, פוּנְקְצִיוֹנָלִי, בְּרִיאוּתִי וּפְּסִיכוֹלוֹגִי.

ש: הַשֵׁנִי.

א: הַצַרְכָן עַצְמוֹ. קֹדֶם כֹּל הָאִישִׁיוּת. צַרְכַן אִימְפּוּלְסִיבִי מַחֲלִיט מַהֵר גַם בְּהַחְלָטוֹת מַשְׁמָעוּתִיוֹת. וְצַרְכַן שֶׁמִתְנַגֵד לַשִׁנוּי מִתְקַשֶׁה לְהַחֲלִיט גַם בְּהַחְלָטוֹת פְּשׁוּטוֹת.

ש: וְיֵשׁ גַם מִשְׁתַנִים דֵמוֹגְרָפִיִים.

א: לְמָשָׁל הַמַצָב הַכַּלְכָּלִי. מִי שֶׁמַצָבוֹ הַכַּלְכָּלִי פָּחוֹת טוֹב, הַסִכּוּן הַכַּלְכָּלִי שֶׁלוֹ גָדוֹל יוֹתֵר, וְלָכֵן גַם הַמְעֹרָבוּת שֶׁלוֹ גְבוֹהָה יוֹתֵר.

ש: הַשְׁלִישִׁי.

א: גוֹרְמֵי הַהֶקְשֵׁר. עֲיֵפוּת, אוֹ לַחַץ זְמַן בִּזְמַן הַקְנִיָה. אֵלֶה מוֹרִידִים אֶת הַמְעֹרָבוּת.

ש: וְיֵשׁ גוֹרֵם רְבִיעִי?

א: אֶפְשָׁר לְצַיֵן אֶת הַתִקְשֹׁרֶת הַשִׁיווּקִית, שֶׁיְכוֹלָה לְהַעֲלוֹת מְעֹרָבוּת. אֲבָל שְׁלֹשֶׁת הָרִאשׁוֹנִים הֵם הָעִקָרִיִים בַּסִפְרוּת הָאָקָדֵמִית.

ש: וּמָה זֶה אִינְטֶרְאַקְצִיָה בֵּין הַגוֹרְמִים?

א: שֶׁהַהַשְׁפָּעָה שֶׁל גוֹרֵם אֶחָד תְלוּיָה בְּגוֹרֵם אַחֵר. דֻגְמָה לַצַרְכָן וּמוּצָר: בְּעִצוּב הַבַּיִת, לְנָשִׁים תִהְיֶה מְעֹרָבוּת גְבוֹהָה יוֹתֵר. בְּמוּצָרִים לְתִפְעוּל הַגִנָה, לִגְבָרִים.

ש: וְדֻגְמָה לְהֶקְשֵׁר וּמוּצָר.

א: בָּלָגָן בַּחֲנוּת. בַּחֲנוּת שֶׁל בִּגְדֵי יֻקְרָה, הַבָּלָגָן מַעֲלֶה עוֹד יוֹתֵר אֶת הַמְעֹרָבוּת. בְּדוּכַן בְּגָדִים בַּשׁוּק, הַבָּלָגָן מוֹרִיד אוֹתָה עוֹד יוֹתֵר. אוֹתוֹ הֶקְשֵׁר, הַשְׁפָּעָה הֲפוּכָה, לְפִי הַמוּצָר.

ש: וְיֵשׁ גַם אִינְטֶרְאַקְצִיָה מְשֻׁלֶשֶׁת.

א: צַרְכַן בְּמַצָב כַּלְכָּלִי יָרוּד יִנְהַג בִּמְעֹרָבוּת גְבוֹהָה גַם בְּדוּכָן בַּשׁוּק. כָּאן שְׁלֹשֶׁת הַגוֹרְמִים פּוֹעֲלִים יַחַד: הַצַרְכָן, הַמוּצָר וְהַהֶקְשֵׁר.

ש: הַנוֹשֵׂא הַשֵׁנִי לַבְּחִינָה: מוֹדֵל עֲשָׂהאֵל. מָה שְׁנֵי הַצִירִים?

א: צִיר אֶחָד: רָמַת הַמְעֹרָבוּת. גְבוֹהָה אוֹ נְמוּכָה. צִיר שֵׁנִי: אֹפִי הַקְנִיָה. הַאִם הַצַרְכָן מְקַבֵּל בְּכֹל קְנִיָה הַחְלָטָה מְחֻדֶשֶׁת, אַחֲרֵי שֶׁהֶעֱרִיךְ שׁוּב אֶת הַחֲלוּפוֹת. אוֹ שֶׁהוּא קוֹנֶה שׁוּב אֶת מָה שֶׁקָנָה בַּפַּעַם הַקוֹדֶמֶת. כְּלוֹמַר, הֶרְגֵל.

ש: וּמִזֶה יוֹצְאִים אַרְבָּעָה דְפוּסִים. מְעֹרָבוּת גְבוֹהָה, וְהַחְלָטָה מְחֻדֶשֶׁת.

א: קַבָּלַת הַחְלָטָה מֻרְכֶּבֶת. הַצַרְכָן רוֹצֶה לִבְחֹן מֵחָדָשׁ אֶת הַחֲלוּפוֹת, יֵשׁ מַשְׁמָעוּת לַהֶבְדֵלִים בֵּינֵיהֶן, וְאֶפְשָׁר לְהִתְרַשֵׁם מֵהֶם לִפְנֵי הַקְנִיָה. הַדֻגְמָה: רְכִישַׁת דִירָה.

ש: מְעֹרָבוּת נְמוּכָה, וְהַחְלָטָה מְחֻדֶשֶׁת.

א: קַבָּלַת הַחְלָטוֹת מֻגְבֶּלֶת. זֶה מַתְאִים לְחִפּוּשׂ מִגְוָן. יֵשׁ לַצַרְכָן עֵרֶךְ בְּגִיווּן וּבַהִתְנַסוּת, בְּעִקָר בְּקָטֵגוֹרְיוֹת שֶׁעוֹנוֹת עַל צֹרֶךְ חוּשִׁי וְרִגְשִׁי. טַעַם, כְּמוֹ גְלִידוֹת. רֵיחַ. אוֹ נִרְאוּת, כְּמוֹ אֲבִיזְרֵי שֵׂעָר. הַהַעֲרָכָה מְהִירָה, וְלָרֹב מִתְבַּצַעַת בִּנְקֻדַת הַמְכִירָה.

ש: מְעֹרָבוּת נְמוּכָה, וּקְנִיָה חוֹזֶרֶת.

א: אִינֶרְצִיָה. הַצַרְכָן חוֹזֵר עַל הַקְנִיָה מִתוֹךְ הֶרְגֵל. הַסִבָּה: בִּדוּל נָמוּךְ בֵּין הַמוּצָרִים, אוֹ קֹשִׁי לְהָבִין אֶת הַהֶבְדֵלִים. לְמָשָׁל חוֹמְרֵי נִקָיוֹן, אוֹ מוּצְרֵי מָזוֹן בְּסִיסִיִים כְּמוֹ חָלָב. אֵין סִכּוּן מַמָשִׁי בְּגִיווּן, אֲבָל גַם אֵין בּוֹ עֵרֶךְ. כֹּל עוֹד הַתוֹצָאָה טוֹבָה, הוּא קוֹנֶה שׁוּב בְּאֹפֶן אוֹטוֹמָטִי.

ש: מְעֹרָבוּת גְבוֹהָה, וּקְנִיָה חוֹזֶרֶת.

א: נֶאֱמָנוּת. כָּאן יֵשׁ סִכּוּן וְאִי וַדָאוּת בְּהַחְלָטָה חֲדָשָׁה, כִּי קָשֶׁה לְהַעֲרִיךְ מֵרֹאשׁ אֶת הַתוֹצָאָה. הַדֻגְמָאוֹת הֵן שֵׁרוּתִים: יִעוּץ פְּסִיכוֹלוֹגִי, יִעוּץ מִשְׁפָּטִי, טִפּוּל יֹפִי.

ש: וְאֵיךְ נוֹצֶרֶת הַנֶאֱמָנוּת?

א: בַּפַּעַם הָרִאשׁוֹנָה הַצַרְכָן קִבֵּל הַחְלָטָה אַחֲרֵי הַעֲרָכָה מַעֲמִיקָה. הוּא חָוָה שְׂבִיעוּת רָצוֹן גְבוֹהָה מְאוֹד, וְלָכֵן הוּא חוֹזֵר עַל אוֹתָה בְּחִירָה. בִּתְנַאי אֶחָד: שֶׁהוּא מַמְשִׁיךְ לִהְיוֹת מְרֻצֶה.

ש: וְזֶה הַצֶמֶד שֶׁהֲכִי קַל לְהִתְבַּלְבֵּל בּוֹ. אִינֶרְצִיָה וְנֶאֱמָנוּת. בִּשְׁתֵיהֶן קוֹנִים שׁוּב אוֹתוֹ דָבָר.

א: נָכוֹן, וְהַהֶבְדֵל הוּא בַּסִבָּה. בְּאִינֶרְצִיָה חוֹזְרִים כִּי לֹא אִכְפַּת מַסְפִּיק: הַמְעֹרָבוּת נְמוּכָה, וְזֶה הֶרְגֵל. בְּנֶאֱמָנוּת חוֹזְרִים כִּי אִכְפַּת מְאוֹד: רוֹצִים לְהִמָנַע מִסִכּוּן, וְיֵשׁ מַשְׁמָעוּת לַקֶשֶׁר עִם נוֹתֵן הַשֵׁרוּת.

ש: יֵשׁ כְּלַל אֶצְבַּע שֶׁעוֹזֵר לְהַבְדִיל?

א: אֶפְשָׁר לִשְׁאֹל מָה יִקְרֶה אִם הַמִתְחָרֶה בַּמִבְצָע. כְּבָר אָמַרְנוּ שֶׁבִּמְעֹרָבוּת גְבוֹהָה הַצַרְכָן פָּחוֹת מֻשְׁפָּע מִמִבְצָעִים וּמִמָה שֶׁקוֹרֶה עַל הַמַדָף. לָכֵן בְּנֶאֱמָנוּת הוּא לֹא יַחְלִיף בְּקַלוּת. בְּאִינֶרְצִיָה, שֶׁהִיא מְעֹרָבוּת נְמוּכָה, קַל יוֹתֵר לְהָזִיז אוֹתוֹ.

ש: וְהַצֶמֶד הַשֵׁנִי: קַבָּלַת הַחְלָטוֹת מֻגְבֶּלֶת מוּל אִינֶרְצִיָה. בִּשְׁתֵיהֶן הַמְעֹרָבוּת נְמוּכָה.

א: הַהֶבְדֵל הוּא אִם יֵשׁ עֵרֶךְ בְּגִיווּן. בְּחִפּוּשׂ מִגְוָן הַצַרְכָן נֶהֱנֶה לְהַחֲלִיף: הַיוֹם טַעַם אֶחָד, מָחָר טַעַם אַחֵר. בְּאִינֶרְצִיָה אֵין לוֹ שׁוּם עִנְיָן לְהַחֲלִיף.

ש: וְאֵיךְ פּוֹתְרִים שְׁאֵלַת מִקְרֶה עַל הַמוֹדֵל?

א: בִּשְׁתֵי שְׁאֵלוֹת. אַחַת: הַמְעֹרָבוּת גְבוֹהָה אוֹ נְמוּכָה? שְׁתַיִם: הַחְלָטָה מְחֻדֶשֶׁת, אוֹ חֲזָרָה עַל אוֹתָה קְנִיָה? שְׁתֵי הַתְשׁוּבוֹת נוֹתְנוֹת אֶת הַתָא בְּמַטְרִיצָה.

ש: וּמָה לֹא נִכְלַל בַּבְּחִינָה, מִתוֹךְ הַמַצֶגֶת הַזֹאת?

א: בְּסִכּוּם כָּתוּב בִּמְפוֹרָשׁ: כֹּל מָה שֶׁבָּא אַחֲרֵי אַרְבַּעַת הַדְפוּסִים הוּא הַעֲשָׁרָה בִּלְבַד, וְאֵינוֹ לַבְּחִינָה. כְּלוֹמַר מַאֲמַצֵי הַשִׁיווּק בְּרָמוֹת מְעֹרָבוּת שׁוֹנוֹת, תֵאוֹרְיַת הַלְמִידָה הַפָּסִיבִית, מוֹדֵל עֶצֶם הַחֲשִׂיפָה, וְשִׁנוּי רָמַת הַמְעֹרָבוּת.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: מָהֵם שְׁלֹשֶׁת הַגוֹרְמִים הָעִקָרִיִים לְרָמַת הַמְעֹרָבוּת?

~4

א: הַמוּצָר, כְּלוֹמַר הַתוֹעָלוֹת וְהַסִכּוּנִים שֶׁבּוֹ. הַצַרְכָן עַצְמוֹ, אִישִׁיוּת וְדֵמוֹגְרַפְיָה. וְגוֹרְמֵי הַהֶקְשֵׁר.

ש: שְׁנִיָה: עֲיֵפוּת וְלַחַץ זְמַן. לְאֵיזֶה גוֹרֵם הֵם שַׁיָכִים, וּלְאֵיזֶה כִּיווּן הֵם מַשְׁפִּיעִים?

~4

א: גוֹרְמֵי הֶקְשֵׁר. הֵם מוֹרִידִים אֶת הַמְעֹרָבוּת.

ש: שְׁלִישִׁית: מָהֵם חֲמֵשֶׁת סוּגֵי הַסִכּוּנִים?

~4

א: כַּלְכָּלִי, חֶבְרָתִי, פוּנְקְצִיוֹנָלִי, בְּרִיאוּתִי וּפְּסִיכוֹלוֹגִי.

ש: רְבִיעִית: לָקוֹחָה חוֹזֶרֶת כְּבָר שָׁנִים לְאוֹתָה עוֹרֶכֶת דִין, כִּי הִיא מְרֻצָה מְאוֹד, וְחוֹשֶׁשֶׁת לְנַסוֹת מִישֶׁהִי אַחֶרֶת.

~4

א: נֶאֱמָנוּת. מְעֹרָבוּת גְבוֹהָה, וּקְנִיָה חוֹזֶרֶת.

ש: חֲמִישִׁית: צַרְכַן לוֹקֵחַ כֹּל שָׁבוּעַ אֶת אוֹתוֹ חָלָב, בְּלִי לַחֲשֹׁב.

~4

א: אִינֶרְצִיָה. מְעֹרָבוּת נְמוּכָה, וּקְנִיָה חוֹזֶרֶת.

ש: שִׁשִׁית: בְּכֹל בִּקוּר בִּגְלִידֵרִיָה הִיא בּוֹחֶרֶת טַעַם אַחֵר.

~4

א: קַבָּלַת הַחְלָטוֹת מֻגְבֶּלֶת. חִפּוּשׂ מִגְוָן.

ש: וְאַחֲרוֹנָה: זוּג מַשְׁוֶה בְּמֶשֶׁךְ חוֹדָשִׁים בֵּין דִירוֹת לִפְנֵי הַקְנִיָה.

~4

א: קַבָּלַת הַחְלָטָה מֻרְכֶּבֶת. מְעֹרָבוּת גְבוֹהָה, וְהַחְלָטָה מְחֻדֶשֶׁת.

ש: זֶהוּ לַפֶּרֶק הַזֶה. בַּפֶּרֶק הַבָּא, הָאַחֲרוֹן: חֲזָרָה לְמִבְחָן. הַמֻשָׂגִים שֶׁמִתְבַּלְבְּלִים, וּשְׁאֵלוֹת בְּסִגְנוֹן שֶׁל דַף הַתִרְגוּל.
