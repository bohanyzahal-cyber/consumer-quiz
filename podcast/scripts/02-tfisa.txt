# יחידה 3. כל עובדה כאן מופיעה בסיכום המצגת ובבנק השאלות src/bank3.js.
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק שֵׁנִי: תְפִיסָה. מַצֶגֶת שָׁלוֹשׁ. שְׁלוֹשָׁה שְׁלַבִּים: חֲשִׂיפָה, קֶשֶׁב וּפַרְשָׁנוּת. נַתְחִיל בַּחֲשִׂיפָה.

א: הַחֲשִׂיפָה הִיא הַקְלִיטָה הָרִאשׁוֹנִית שֶׁל הַגֵרוּי בַּחוּשִׁים. רְאִיָה, שְׁמִיעָה, טַעַם, רֵיחַ, מִשׁוּשׁ, תְנוּעָה וְהִתְמַצְאוּת בַּמֶרְחָב. בַּשָׁלָב הַזֶה הַגֵרוּי נִקְלָט בַּצוּרָה הַגוֹלְמִית שֶׁלוֹ: צְלִיל, קַו, אוֹת. עֲדַיִן בְּלִי מַשְׁמָעוּת.

ש: וְלָמָה כְּדַאי לְהַפְעִיל כַּמָה חוּשִׁים יַחַד?

א: כִּי כֹּל חוּשׁ קוֹלֵט בְּנִפְרָד אֶת הַחֵלֶק שֶׁנוֹגֵעַ אֵלָיו. אִם אֲנִי רוֹאֶה אֶת הַמִלָה טָרִי, וְגַם שׁוֹמֵעַ אוֹתָה, הַגֵרוּי נִקְלָט דֶרֶךְ יוֹתֵר עֲרוּצִים.

ש: וְיֵשׁ גַם עִנְיָן שֶׁל שְׁנֵי חֲצָאֵי הַמוֹחַ.

א: מֵידָע חָזוּתִי מַפְעִיל אֶת הַחֵצִי הַיְמָנִי. מֵידָע עֻבְדָתִי אוֹ מִלוּלִי מַפְעִיל אֶת הַחֵצִי הַשְׂמָאלִי. כְּשֶׁמְשַׁלְבִים בֵּינֵיהֶם, נוֹצַר גֵרוּי שֶׁיֵשׁ לוֹ קְשָׁרִים חֲזָקִים יוֹתֵר.

ש: עַכְשָׁו שְׁנֵי סִפִּים, שֶׁחַיָבִים לְהַבְדִיל בֵּינֵיהֶם. הָרִאשׁוֹן.

א: הַסַף הַמֻחְלָט. עוֹצְמַת הַגֵרוּי הַמִינִימָלִית שֶׁנִדְרֶשֶׁת כְּדֵי שֶׁנִקְלֹט אֶת הַגֵרוּי בַּחוּשִׁים.

ש: וּמָה קוֹרֶה מִתַחַת לַסַף הַזֶה?

א: גֵרוּי שֶׁהָעוֹצְמָה שֶׁלוֹ נְמוּכָה מֵהַסַף הַמֻחְלָט נִקְרָא גֵרוּי תַת סִפִּי. הוּא יָכוֹל לְהִקָלֵט בַּחוּשִׁים, וְאוּלַי אֲפִלוּ לְהַשְׁפִּיעַ, אֲבָל אֲנַחְנוּ לֹא מוּדָעִים אֵלָיו.

ש: זֹאת הַשְׁאֵלָה עַל הָרֵיחַ בַּכְּנִיסָה לַחֲנוּת, בְּדַף הַתִרְגוּל.

א: בְּדִיוּק. רֵיחַ בְּעוֹצְמָה נְמוּכָה מְאוֹד, הַלָקוֹחוֹת לֹא מוּדָעִים אֵלָיו, וְאוּלַי הוּא בְּכֹל זֹאת מַשְׁפִּיעַ. הַתְשׁוּבָה: גֵרוּי תַת סִפִּי.

ש: וְהַסַף הַשֵׁנִי.

א: סַף הַהַבְחָנָה בַּהֶבְדֵל. כָּאן לֹא שׁוֹאֲלִים אִם קוֹלְטִים אֶת הַגֵרוּי, אֶלָא אִם מַרְגִישִׁים שֶׁהוּא הִשְׁתַנָה. כְּדֵי שֶׁנַרְגִישׁ בְּשִׁנוּי, הוּא צָרִיךְ לִהְיוֹת בְּעוֹצְמָה מַסְפֶּקֶת.

ש: וְכָאן נִכְנָס חֹק וֶבֶּר.

א: חֹק וֶבֶּר אוֹמֵר: כְּכֹל שֶׁהַגֵרוּי הָרִאשׁוֹנִי גָדוֹל יוֹתֵר, צָרִיךְ שִׁנוּי גָדוֹל יוֹתֵר כְּדֵי שֶׁנַבְחִין בּוֹ. תוֹסֶפֶת שֶׁל שְׁנֵי שְׁקָלִים מֻרְגֶשֶׁת מְאוֹד בְּמוּצָר שֶׁל עֲשָׂרָה שְׁקָלִים, וְכִמְעַט לֹא מֻרְגֶשֶׁת בְּמוּצָר שֶׁל מָאתַיִם.

ש: וּמָה הַמְשַׁוֶקֶת עוֹשָׂה עִם זֶה?

א: הִיא מַחֲלִיטָה מָה הִיא רוֹצָה. אִם הַשִׁנוּי מֵיטִיב עִם הַצַרְכָן, הִיא רוֹצָה שֶׁהוּא יַרְגִישׁ בּוֹ, וְלָכֵן הַשִׁנוּי צָרִיךְ לַעֲבֹר אֶת הַסַף. אִם לֹא, אוּלַי הִיא מַעֲדִיפָה שֶׁהַשִׁנוּי לֹא יֻרְגַשׁ.

ש: וְיֵשׁ כָּאן גַם צַד רֵגוּלָטוֹרִי.

א: כֵּן. אִם שִׁנוּ אֶת תְנָאֵי הַמוּצָר לְרָעַת הַצַרְכָן, לְמָשָׁל הַמְחִיר עָלָה, צָרִיךְ לִכְתֹב אֶת זֶה בְּצוּרָה בְּרוּרָה עַל הַמוּצָר אוֹ עַל הָאֲרִיזָה. אֲבָל לְפִי הַסִכּוּם, לֹא כֹּל הַמְשַׁוְקִים מַקְפִּידִים, וְהַנוֹשֵׂא לֹא נֶאֱכַף בִּקְפִידָה.

ש: הַשָׁלָב הַשֵׁנִי: קֶשֶׁב.

א: קֶשֶׁב הוּא הַהַחְלָטָה בְּאֵלוּ גֵרוּיִים לְמַקֵד אֶת תְשׂוּמֶת הַלֵב. הַצַרְכָנִים חֲשׂוּפִים לְאַלְפֵי מְסָרִים, וְהַמְשַׁוֵק רוֹצֶה לְכַוֵן אֶת הַקֶשֶׁב אֶל הַגֵרוּי שֶׁלוֹ. יֵשׁ לְזֶה שְׁתֵי דְרָכִים.

ש: הָרִאשׁוֹנָה.

א: קֶשֶׁב שֶׁמְקוֹרוֹ בְּגֵרוּי. הוּא נִקְרָא גַם קֶשֶׁב לֹא רְצוֹנִי, אוֹ בּוֹטוֹם אַפּ. הַגֵרוּי בּוֹלֵט, חָזָק מֵהָרָגִיל, יוֹצֵא דֹפֶן אוֹ לֹא צָפוּי. לְמָשָׁל מוּצָר בִּתְצוּגָה שֶׁזָזָה, כְּשֶׁכֹּל שְׁאָר הַתְצוּגָה עוֹמֶדֶת.

ש: וְהַשְׁנִיָה.

א: קֶשֶׁב שֶׁמְקוֹרוֹ בְּצַרְכָן. קֶשֶׁב רְצוֹנִי, טוֹפּ דָאוּן. כָּאן הַגֵרוּי פּוֹנֶה לַצְרָכִים שֶׁל הַצַרְכָן, וּלְנוֹשְׂאִים שֶׁחֲשׁוּבִים לוֹ. וְזֶה יוֹצֵר אֶצְלוֹ עֵרָנוּת תְפִיסָתִית.

ש: אֵיךְ מַבְדִילִים בֵּינֵיהֶם בַּשְׁאֵלָה?

א: שׁוֹאֲלִים: מָה מָשַׁךְ אֶת תְשׂוּמֶת הַלֵב? אִם זֹאת הַבּוֹלְטוֹת שֶׁל הַגֵרוּי עַצְמוֹ, צֶבַע, תְנוּעָה, צְלִיל, זֶה קֶשֶׁב לֹא רְצוֹנִי. אִם הַגֵרוּי דווקא רָגִיל, אֲבָל נוֹגֵעַ לְמַשֶׁהוּ שֶׁמַעֲסִיק אֶת הַצַרְכָן, זֶה קֶשֶׁב רְצוֹנִי.

ש: וְזֹאת הַשְׁאֵלָה עַל הַמַצְלֵמָה.

א: כֵּן. צַרְכָנִית מְחַפֶּשֶׂת שָׁבוּעַ מַצְלֵמָה לְטִיוּל. וְאָז הִיא מַבְחִינָה מִיָד בְּמוֹדָעָה פְּשׁוּטָה לַמַצְלֵמָה, בְּלִי צֶבַע חָרִיג, בְּלִי תְנוּעָה וּבְלִי צְלִיל. הַמוֹדָעָה לֹא בּוֹלֶטֶת, אֲבָל הִיא רֵלֵוַנְטִית לַצֹרֶךְ. לָכֵן: קֶשֶׁב רְצוֹנִי, טוֹפּ דָאוּן.

ש: וְיֵשׁ דֻגְמָה דוֹמָה בְּסִכּוּם, עַל ווֹלְט.

א: לִקְרַאת פֶּסַח, ווֹלְט אִפְשְׁרָה לִתְרֹם מַאַרְזֵי מָזוֹן לְנִזְקָקִים. הַטַעֲנָה שֶׁזֶה קֶשֶׁב שֶׁמְקוֹרוֹ בְּגֵרוּי: לֹא נְכוֹנָה. זֶה קֶשֶׁב שֶׁמְקוֹרוֹ בְּצַרְכָן, כִּי הַמַהֲלָךְ פּוֹנֶה לְנוֹשְׂאִים שֶׁקְרוֹבִים לַלֵב, כְּמוֹ תְרוּמָה לַקְהִלָה, וּבַהֶקְשֵׁר שֶׁל הַחַג.

ש: הַשָׁלָב הַשְׁלִישִׁי: פַּרְשָׁנוּת.

א: פַּרְשָׁנוּת הִיא הָאֹפֶן שֶׁבּוֹ הָאָדָם מְאַרְגֵן אֶת הָרְשָׁמִים שֶׁקִבֵּל מֵהַחוּשִׁים, מְזַהֶה אוֹתָם, וְנוֹתֵן לָהֶם מַשְׁמָעוּת. וּבָזֶה עוֹסֶקֶת אַסְכּוֹלַת הַגֶשְׁטַלְט.

ש: מָה הַמִשְׁפָּט שֶׁמְסַכֵּם אֶת הַגֶשְׁטַלְט?

א: הַשָׁלֵם שׁוֹנֶה מִסְכוּם חֲלָקָיו.

ש: וּמֵאֵיפֹה זֶה הִתְחִיל?

א: מֵהַנִסוּי שֶׁל וֶרְטְהַיְמֶר. הוּא הִקְרִין שְׁנֵי הֶבְזֵקֵי אוֹר, מִשְׁתֵי נְקֻדוֹת שׁוֹנוֹת, בְּהֶפְרֵשׁ זְמַן קָצָר. הַנִבְדָקִים רָאוּ נְקֻדַת אוֹר אַחַת שֶׁנָעָה. אַף שֶׁלֹא הָיְתָה שׁוּם תְנוּעָה. הַמַסְקָנָה: הַתוֹדָעָה מַשְׁלִיכָה תַבְנִית עַל מָה שֶׁהַחוּשִׁים קוֹלְטִים.

ש: שִׁשָׁה חֻקִים. הָרִאשׁוֹן, וְהַיְסוֹדִי בְּיוֹתֵר.

א: דְמוּת וְרֶקַע. הָאָדָם מַבְחִין בֵּין הַדְמוּת, שֶׁהִיא עִקַר הַקֶשֶׁב, לְבֵין הָרֶקַע, שֶׁהוּא הַמֵידָע הַמִשְׁנִי. הַמְשַׁוֵק מַחֲלִיט מָה הַמֶסֶר הָעִקָרִי, וּמַבְלִיט אוֹתוֹ כִּדְמוּת.

ש: כְּמוֹ בַּשְׁאֵלָה עַל כַּפְתוֹר הַתַשְׁלוּם.

א: כַּפְתוֹר גָדוֹל, מְבֻדָד וּבָרוּר, וְהַמֵידָע הַמִשְׁנִי פָּחוֹת בּוֹלֵט סְבִיבוֹ. זֶה דְמוּת וְרֶקַע.

ש: וּמָה קוֹרֶה כְּשֶׁמְטַשְׁטְשִׁים בֵּין הַדְמוּת לָרֶקַע?

א: זֶה מַכְבִּיד עַל עִבּוּד הַמֵידָע. אֲבָל לִפְעָמִים דווקא הַמַאֲמָץ לְהַחֲלִיט בְּמָה לְהִתְמַקֵד יוֹצֵר מְעֹרָבוּת גְבוֹהָה יוֹתֵר, וְעִבּוּד עָמֹק יוֹתֵר.

ש: הַחֹק הַשֵׁנִי.

א: סְגִירוּת, אוֹ הַשְׁלָמָה. גַם כְּשֶׁהַמֵידָע לֹא שָׁלֵם, אֲנַחְנוּ מַשְׁלִימִים אוֹתוֹ לְתַבְנִית עִם גְבוּלוֹת בְּרוּרִים. וְהָרָצוֹן לְהַשְׁלִים יוֹצֵר מְעֹרָבוּת. לְמָשָׁל לוֹגוֹ שֶׁמְגָרֶה לְהַשְׁלִים, טִיזֵר, אוֹ סִפּוּר שֶׁמְעוֹרֵר סַקְרָנוּת.

ש: הַשְׁלִישִׁי.

א: קִרְבָה, אוֹ סְמִיכוּת. דְבָרִים שֶׁקְרוֹבִים בַּזְמַן וּבַמֶרְחָב נִתְפָּסִים כִּקְשׁוּרִים. הַדֻגְמָה: נַעֲלֵי בַּד לְיַד נַעֲלֵי נוֹחוּת נִתְפָּסוֹת כְּנַעֲלַיִם לְיוֹם יוֹם. לְיַד גַרְבַּיִם, הֵן נִתְפָּסוֹת כְּנַעֲלֵי בַּיִת.

ש: הָרְבִיעִי.

א: דִמְיוֹן. אֵלֵמֶנְטִים עִם מְאַפְיְנִים חָזוּתִיִים דוֹמִים נִתְפָּסִים כִּקְשׁוּרִים. לְמָשָׁל אֲתָר שֶׁבּוֹ הַפְּרִיטִים מֵאוֹתוֹ תְחוּם מְעֻצָבִים בְּאוֹתוֹ אֹפֶן.

ש: וְזֶה צֶמֶד שֶׁקַל לְהִתְבַּלְבֵּל בּוֹ. קִרְבָה וְדִמְיוֹן.

א: בְּקִרְבָה, שָׂמִים דְבָרִים אֶחָד לְיַד הַשֵׁנִי. בַּדִמְיוֹן, הֵם נִרְאִים אוֹתוֹ דָבָר. בַּסִכּוּם יֵשׁ שְׁאֵלָה: חֲטִיף בְּרִיאוּת שֶׁמֻצָב לְיַד אֱגוֹזִים וּפֵרוֹת. זֶה לֹא דִמְיוֹן. זֹאת סְמִיכוּת. וּלְעֻמַת זֶה, קוֹלֶקְצִיָה שֶׁל זָארָה שֶׁמְעֻצֶבֶת בְּאַפְּלִיקַצְיָה כְּמוֹ יְצִירוֹת אוֹמָנוּת: זֶה דִמְיוֹן.

ש: הַחֲמִישִׁי.

א: רְצִיפוּת טוֹבָה, אוֹ הֶמְשֵׁכִיוּת. אֵלֵמֶנְטִים שֶׁמְסֻדָרִים עַל קַו אוֹ עַל עֲקֻמָה נִתְפָּסִים כִּקְשׁוּרִים.

ש: וְהַשִׁשִׁי.

א: פַּשְׁטוּת. הַמוֹחַ מַעֲדִיף דְבָרִים פְּשׁוּטִים, בְּרוּרִים וּמְסֻדָרִים. הֵם נִתְפָּסִים כִּבְטוּחִים יוֹתֵר, וְדוֹרְשִׁים מִמֶנוּ פָּחוֹת זְמַן וּמַשְׁאַבִּים.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: מָה הַהֶבְדֵל בֵּין הַסַף הַמֻחְלָט לְסַף הַהַבְחָנָה בַּהֶבְדֵל?

~4

א: הַסַף הַמֻחְלָט הוּא הָעוֹצְמָה הַמִינִימָלִית כְּדֵי לִקְלֹט גֵרוּי. סַף הַהַבְחָנָה הוּא הַשִׁנוּי הַמִינִימָלִי כְּדֵי לְהַרְגִישׁ שֶׁהַגֵרוּי הִשְׁתַנָה.

ש: שְׁנִיָה: מָה אוֹמֵר חֹק וֶבֶּר?

~4

א: כְּכֹל שֶׁהַגֵרוּי הָרִאשׁוֹנִי גָדוֹל יוֹתֵר, צָרִיךְ שִׁנוּי גָדוֹל יוֹתֵר כְּדֵי לְהַבְחִין בּוֹ.

ש: שְׁלִישִׁית: מוֹדָעָה אֲפֹרָה וּרְגִילָה, שֶׁהַצַרְכָן שָׂם לֵב אֵלֶיהָ כִּי הִיא עוֹסֶקֶת בְּדִיוּק בְּמָה שֶׁהוּא מְחַפֵּשׂ. אֵיזֶה קֶשֶׁב זֶה?

~4

א: קֶשֶׁב רְצוֹנִי, שֶׁמְקוֹרוֹ בְּצַרְכָן. טוֹפּ דָאוּן.

ש: רְבִיעִית: חֲטִיף בְּרִיאוּת שֶׁמֻצָב לְיַד פֵּרוֹת. אֵיזֶה חֹק גֶשְׁטַלְט?

~4

א: קִרְבָה, כְּלוֹמַר סְמִיכוּת. לֹא דִמְיוֹן.

ש: חֲמִישִׁית: לוֹגוֹ שֶׁחָסֵר בּוֹ חֵלֶק, וְהַצוֹפֶה מַשְׁלִים אוֹתוֹ בְּעַצְמוֹ.

~4

א: סְגִירוּת.

ש: וְאַחֲרוֹנָה: מָה הַמִשְׁפָּט שֶׁל הַגֶשְׁטַלְט?

~4

א: הַשָׁלֵם שׁוֹנֶה מִסְכוּם חֲלָקָיו.

ש: זֶהוּ לַפֶּרֶק הַזֶה. בַּפֶּרֶק הַבָּא: הַזִכָּרוֹן הַמְפוֹרָשׁ. טווח קָצָר, טווח אָרֹךְ, הָרֶשֶׁת הַסֵמַנְטִית וְהַחֲסִימוֹת.
