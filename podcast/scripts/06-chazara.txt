# title: חזרה למבחן — מושגים שמתבלבלים ושאלות בסגנון המרצה
# חזרה על יחידות 2 עד 5. כל עובדה כאן מופיעה בפרקים הקודמים ובבנק השאלות.
# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק שִׁשִׁי וְאַחֲרוֹן: חֲזָרָה לְמִבְחָן. נַתְחִיל מִמָה שֶׁיָדוּעַ עַל הַבְּחִינָה עַצְמָה.

א: לְפִי דַף הַתִרְגוּל שֶׁקִבַּלְתֶם: אַרְבַּע עֶשְׂרֵה שְׁאֵלוֹת אָמֵרִיקָאִיוֹת, וּמֻתָר לְהַכְנִיס חֹמֶר עֵזֶר מֻדְפָּס. בִּשְׁאֵלוֹת הַתִרְגוּל שֶׁבַּדַף יֵשׁ חָמֵשׁ תְשׁוּבוֹת לְכֹל שְׁאֵלָה, אַחַת נְכוֹנָה, וְהַסֵדֶר מְעֻרְבָּל.

ש: וּמָה הַנוֹשְׂאִים?

א: אַרְבָּעָה. יְחִידָה שְׁתַיִם: הַכָּרָה בַּצֹרֶךְ, וּמוֹדֵל הָעֲרָכִים לְלָקוֹחוֹת פְּרָטִיִים וְעִסְקִיִים. מַצֶגֶת שָׁלוֹשׁ: תְפִיסָה. חֲשִׂיפָה, קֶשֶׁב, וּפַרְשָׁנוּת לְפִי הַגֶשְׁטַלְט. מַצֶגֶת אַרְבַּע: זִכָּרוֹן. טווח קָצָר, טווח אָרֹךְ וְהָרֶשֶׁת הַסֵמַנְטִית, חֲסִימוֹת, וּמְרֻמָז מוּל מְפוֹרָשׁ. וּמַצֶגֶת חָמֵשׁ: מְעֹרָבוּת. הַגוֹרְמִים לִמְעֹרָבוּת, וּדְפוּסֵי הַקְנִיָה לְפִי עֲשָׂהאֵל.

ש: וְאֵיךְ נִרְאוֹת הַשְׁאֵלוֹת שֶׁבַּדַף?

א: כִּמְעַט כֻּלָן בְּנוּיוֹת אוֹתוֹ דָבָר. מִקְרֶה קָצָר מֵהַחַיִים, וְאָז צָרִיךְ לְזַהוֹת אֵיזֶה מֻשָׂג הוּא מַדְגִים. לָכֵן לֹא מַסְפִּיק לָדַעַת הַגְדָרוֹת. צָרִיךְ לְזַהוֹת אֶת הַמֻשָׂג בְּתוֹךְ סִפּוּר.

ש: וְאֵיךְ בְּנוּיִים הַמַסִיחִים?

א: הַרְבֵּה מֵהֶם הֵם מֻשָׂגִים אֲמִתִיִים, אֲבָל מִפֶּרֶק אַחֵר. בַּשְׁאֵלָה עַל קֶשֶׁב מוֹפִיעָה חֲסִימָה לְפָנִים. בַּשְׁאֵלָה עַל גֶשְׁטַלְט מוֹפִיעָה חֲסִימָה לְאָחוֹר. אָז הַשָׁלָב הָרִאשׁוֹן הוּא לִשְׁאֹל: עַל אֵיזֶה פֶּרֶק הַשְׁאֵלָה? וְזֶה מוֹחֵק מִיָד שְׁתַיִם אוֹ שָׁלוֹשׁ תְשׁוּבוֹת.

ש: וְעוֹד סִימָן?

א: מִלִים מֻחְלָטוֹת. בִּלְבַד, תָמִיד, בְּהֶכְרֵחַ, רַק. בְּדַף הַתִרְגוּל הֵן מוֹפִיעוֹת בְּעִקָר בַּתְשׁוּבוֹת הַלֹא נְכוֹנוֹת. זֶה לֹא חֹק, אֲבָל זֶה סִימָן שֶׁשָׁוֶה לַעֲצֹר עָלָיו.

ש: וּמָה עוֹשִׂים עִם חֹמֶר הָעֵזֶר?

א: מְכִינִים דַף מְסֻדָר מֵרֹאשׁ, עִם הַהַבְחָנוֹת וְהָרְשִׁימוֹת. בַּבְּחִינָה אֵין זְמַן לְחַפֵּשׂ בְּתוֹךְ סִכּוּמִים אֲרֻכִּים. בַּאֲתַר הַתִרְגוּל יֵשׁ דַף עֵזֶר מוּכָן לְהַדְפָּסָה.

ש: עַכְשָׁו עֲשָׂרָה צְמָדִים שֶׁמִתְבַּלְבְּלִים. אֲנִי אוֹמֶרֶת אֶת הַצֶמֶד, וְאַתָה אֶת הַהֶבְדֵל. רִאשׁוֹן: צֹרֶךְ וְרָצוֹן.

א: הַצֹרֶךְ הוּא הַפַּעַר בֵּין מָצוּי לְרָצוּי, יַעַד עָמֹק וּמֻפְשָׁט. הָרָצוֹן הוּא הַדֶרֶךְ הַמְסֻיֶמֶת לְסַפֵּק אוֹתוֹ. הַמְשַׁוֶקֶת לֹא יוֹצֶרֶת צֹרֶךְ. הִיא מְעוֹרֶרֶת רָצוֹן.

ש: שֵׁנִי: דְרִישַׁת סַף וְעֵרֶךְ פוּנְקְצִיוֹנָלִי, בַּפִּירָמִידָה שֶׁל הַלָקוֹחוֹת הָעִסְקִיִים.

א: דְרִישַׁת סַף הִיא תְנַאי כְּנִיסָה: רֵגוּלַצְיָה, אַבְטָחָה, דְרִישׁוֹת מִשְׁפָּטִיוֹת, אֵתִיוֹת אוֹ טֶכְנִיוֹת. עֵרֶךְ פוּנְקְצִיוֹנָלִי הוּא מָה שֶׁהַמוּצָר נוֹתֵן לָעֵסֶק. לְמָשָׁל הַיְכֹלֶת לִגְדֹל יַחַד אִתוֹ.

ש: שְׁלִישִׁי: הַסַף הַמֻחְלָט וְסַף הַהַבְחָנָה בַּהֶבְדֵל.

א: הַסַף הַמֻחְלָט: הַאִם בִּכְלָל קוֹלְטִים אֶת הַגֵרוּי. סַף הַהַבְחָנָה: הַאִם מַרְגִישִׁים שֶׁהוּא הִשְׁתַנָה. חֹק וֶבֶּר שַׁיָךְ לְסַף הַהַבְחָנָה. וְגֵרוּי תַת סִפִּי שַׁיָךְ לַסַף הַמֻחְלָט.

ש: רְבִיעִי: קֶשֶׁב רְצוֹנִי וְקֶשֶׁב לֹא רְצוֹנִי.

א: לֹא רְצוֹנִי: הַגֵרוּי עַצְמוֹ בּוֹלֵט. צֶבַע, תְנוּעָה, מַשֶׁהוּ לֹא צָפוּי. זֶה בּוֹטוֹם אַפּ. רְצוֹנִי: הַגֵרוּי נוֹגֵעַ לְמַשֶׁהוּ שֶׁמַעֲסִיק אֶת הַצַרְכָן. זֶה טוֹפּ דָאוּן.

ש: חֲמִישִׁי: קִרְבָה וְדִמְיוֹן.

א: קִרְבָה: הַדְבָרִים מֻנָחִים זֶה לְיַד זֶה. דִמְיוֹן: הֵם נִרְאִים אוֹתוֹ דָבָר.

ש: שִׁשִׁי: דְמוּת וְרֶקַע, מוּל סְגִירוּת.

א: דְמוּת וְרֶקַע: מַבְלִיטִים דָבָר אֶחָד, וְכֹל הַשְׁאָר נָסוֹג לָרֶקַע. סְגִירוּת: חָסֵר מַשֶׁהוּ, וְהַצוֹפֶה מַשְׁלִים אוֹתוֹ בְּעַצְמוֹ.

ש: שְׁבִיעִי: הַתָקָה וּדְעִיכָה.

א: שְׁתֵיהֶן שִׁכְחָה בַּזִכָּרוֹן לטווח קָצָר. הַתָקָה: פְּרִיט חָדָשׁ תָפַס אֶת הַמָקוֹם. דְעִיכָה: פָּשׁוּט עָבַר זְמַן.

ש: שְׁמִינִי: הַתָקָה וַחֲסִימָה לְאָחוֹר.

א: בִּשְׁתֵיהֶן חָדָשׁ פּוֹגֵעַ בַּיָשָׁן. הַתָקָה בַּזִכָּרוֹן לטווח קָצָר. חֲסִימָה לְאָחוֹר בְּזִכָּרוֹן לטווח אָרֹךְ.

ש: תְשִׁיעִי: חֲסִימָה לְאָחוֹר וַחֲסִימָה לְפָנִים.

א: שׁוֹאֲלִים מָה נִפְגַע. הַיָשָׁן נִשְׁכַּח: חֲסִימָה לְאָחוֹר. הֶחָדָשׁ לֹא נִקְלַט: חֲסִימָה לְפָנִים.

ש: עֲשִׂירִי: צְפִיפוּת וְחֹזֶק בָּרֶשֶׁת הַסֵמַנְטִית.

א: צְפִיפוּת: כַּמָה קְשָׁרִים יֵשׁ לַמֻשָׂג. מוֹסִיפִים עֲרָכִים, וְהַצְפִיפוּת גְדֵלָה. חֹזֶק: כַּמָה חָזָק כֹּל קֶשֶׁר. וְהוּא נִבְנָה מֵחֲזָרָה עַל אוֹתוֹ עֵרֶךְ.

ש: וְעוֹד אֶחָד לְסִיוּם: אִינֶרְצִיָה וְנֶאֱמָנוּת.

א: בִּשְׁתֵיהֶן קוֹנִים שׁוּב אוֹתוֹ דָבָר. בָּאִינֶרְצִיָה הַמְעֹרָבוּת נְמוּכָה, וְזֶה הֶרְגֵל. בְּנֶאֱמָנוּת הַמְעֹרָבוּת גְבוֹהָה, וְנִמְנָעִים מִסִכּוּן.

ש: עַכְשָׁו תִרְגוּל. אֲנִי מְתָאֶרֶת מִקְרֶה, יֵשׁ כַּמָה שְׁנִיוֹת לַחֲשֹׁב, וְאָז הַתְשׁוּבָה. מוּכָנִים? מִקְרֶה רִאשׁוֹן. יַצְרָנִית שׁוֹקוֹלָד הִקְטִינָה אֶת הַחֲפִיסָה מִמֵאָה גְרָם לְתִשְׁעִים וַחֲמִשָׁה, וְהַלָקוֹחוֹת לֹא שָׂמוּ לֵב.

~5

א: סַף הַהַבְחָנָה בַּהֶבְדֵל. הַשִׁנוּי קָטָן מִדַי בְּיַחַס לַגֵרוּי הַמְקוֹרִי. זֶה חֹק וֶבֶּר.

ש: מִקְרֶה שֵׁנִי. שֶׁלֶט חוּצוֹת שֶׁבּוֹ דְמוּת זָזָה, בֵּין עֶשְׂרוֹת שְׁלָטִים שֶׁעוֹמְדִים בַּמָקוֹם.

~5

א: קֶשֶׁב לֹא רְצוֹנִי, שֶׁמְקוֹרוֹ בְּגֵרוּי. בּוֹטוֹם אַפּ.

ש: מִקְרֶה שְׁלִישִׁי. חֲנוּת מַצִיבָה מִטְרִיוֹת לְיַד מְעִילֵי גֶשֶׁם, וְהַלָקוֹחוֹת תוֹפְסִים אוֹתָם כְּסֵט אֶחָד.

~5

א: חֹק הַקִרְבָה. סְמִיכוּת.

ש: מִקְרֶה רְבִיעִי. פִּרְסֹמֶת שֶׁנִקְטַעַת בְּאֶמְצַע הַמִשְׁפָּט, וְהַצוֹפִים מַשְׁלִימִים אוֹתוֹ בָּרֹאשׁ.

~5

א: סְגִירוּת.

ש: מִקְרֶה חֲמִישִׁי. מוֹכֵר מַקְרִיא לַלָקוֹחָה שְׁתֵים עֶשְׂרֵה תְכוּנוֹת שֶׁל מוּצָר בְּרֶצֶף, וְהִיא זוֹכֶרֶת רַק חֵלֶק.

~5

א: מִגְבֶּלֶת הַקִבֹּלֶת שֶׁל הַזִכָּרוֹן לטווח קָצָר. חֲמִשָׁה עַד תִשְׁעָה גוּשֵׁי מֵידָע. הַפִּתְרוֹן: לְאַרְגֵן אֶת הַמֵידָע בִּמְעַט גוּשִׁים.

ש: מִקְרֶה שִׁשִׁי. מוּתָג מְזֹהֶה עִם אֵיכוּת, וְעַכְשָׁו גַם עִם קַיָמוּת וְגַם עִם עִצוּב.

~5

א: צְפִיפוּת הַקְשָׁרִים גְדֵלָה. לֹא בְּהֶכְרֵחַ הַחֹזֶק.

ש: מִקְרֶה שְׁבִיעִי. כְּשֶׁשׁוֹאֲלִים עַל מִשְׁלוֹחֵי אֹכֶל, כֻּלָם אוֹמְרִים שֵׁם שֶׁל מוּתָג אֶחָד. וּכְשֶׁאוֹמְרִים אֶת שֵׁם הַמוּתָג, כֻּלָם חוֹשְׁבִים עַל מִשְׁלוֹחֵי אֹכֶל.

~5

א: קֶשֶׁר סִימֶטְרִי.

ש: מִקְרֶה שְׁמִינִי. בַּנְק הֶחְלִיף אֶת הַשֵׁם שֶׁלוֹ, וְהַלָקוֹחוֹת מַמְשִׁיכִים לִקְרֹא לוֹ בַּשֵׁם הַיָשָׁן, וְלֹא מַצְלִיחִים לִזְכֹּר אֶת הֶחָדָשׁ.

~5

א: חֲסִימָה לְפָנִים. הַיֶדַע הַקוֹדֵם מַקְשֶׁה לִזְכֹּר אֶת הֶחָדָשׁ.

ש: מִקְרֶה תְשִׁיעִי. אוֹתוֹ קָפֶה בְּדִיוּק מֻגָשׁ בְּכוֹס קַרְטוֹן, וּבְכוֹס חַרְסִינָה מְעֻצֶבֶת. בְּכוֹס הַחַרְסִינָה הוּא מְקַבֵּל צִיוּן גָבוֹהַ יוֹתֵר, וְהַטוֹעֲמִים בְּטוּחִים שֶׁהַכּוֹס לֹא הִשְׁפִּיעָה.

~5

א: מִסְגוּר, וְזִכָּרוֹן מְרֻמָז.

ש: מִקְרֶה עֲשִׂירִי. חֲנוּת מַצִיגָה לְיַד הַשָׁעוֹן שֶׁהִיא רוֹצָה לִמְכֹּר שָׁעוֹן דוֹמֶה, בִּמְחִיר כָּפוּל.

~5

א: הַקְטָנַת הָעֲלוּת הַנִתְפֶּסֶת, דֶרֶךְ הַהֶקְשֵׁר שֶׁל הַמְחִיר. עֶקְרוֹן הָעֹנֶג וְהַסֵבֶל.

ש: מִקְרֶה אַחַת עֶשְׂרֵה. בַּקָפֵטֶרְיָה שָׂמִים אֶת הַפֵּרוֹת בְּגֹבַה הָעֵינַיִם, וְאֶת הַמַמְתַקִים בַּמַדָף הַתַחְתוֹן. לֹא אוֹסְרִים כְּלוּם, וְלֹא מְשַׁנִים מְחִירִים.

~5

א: דְחִיפָה קַלָה. הַכְוָנָה מְרֻמֶזֶת.

ש: מִקְרֶה שְׁתֵים עֶשְׂרֵה. צַרְכַן עָיֵף מְאוֹד, שֶׁמְמַהֵר לִסְגֹר קְנִיָה לִפְנֵי שֶׁהַחֲנוּת נִסְגֶרֶת.

~5

א: גוֹרְמֵי הֶקְשֵׁר, שֶׁמוֹרִידִים אֶת הַמְעֹרָבוּת.

ש: מִקְרֶה שְׁלוֹשׁ עֶשְׂרֵה. סְטוּדֶנְטִית עִם תַקְצִיב מְצֻמְצָם מַשְׁקִיעָה שָׁעוֹת בְּהַשְׁוָאָה, גַם לִפְנֵי קְנִיָה קְטַנָה.

~5

א: הַצַרְכָן עַצְמוֹ. מַצָב כַּלְכָּלִי פָּחוֹת טוֹב מַגְדִיל אֶת הַסִכּוּן הַכַּלְכָּלִי, וּמַעֲלֶה אֶת הַמְעֹרָבוּת.

ש: מִקְרֶה אַרְבַּע עֶשְׂרֵה. לָקוֹחַ קוֹנֶה תָמִיד אֶת אוֹתוֹ נוֹזֵל כֵּלִים, כִּי הֵם כֻּלָם אוֹתוֹ דָבָר בְּעֵינָיו.

~5

א: אִינֶרְצִיָה.

ש: מִקְרֶה חֲמֵשׁ עֶשְׂרֵה. סַפָּק מַדְגִישׁ לְלָקוֹחַ עִסְקִי שֶׁהַמַעֲרֶכֶת שֶׁלוֹ תַעֲזֹר לִמְנַהֶלֶת הָרֶכֶשׁ לְהִתְקַדֵם בַּקַרְיֵרָה.

~5

א: עֵרֶךְ אִישִׁי, בְּפִּירָמִידַת הָעֲרָכִים שֶׁל הַלָקוֹחוֹת הָעִסְקִיִים.

ש: וְאַחֲרוֹן. חֶבְרָה בּוֹנָה לְכֹל אֶחָד מִשְׁלוֹשָׁה אִרְגוּנֵי יַעַד מֶסֶר נִפְרָד, וְעַמוּד נְחִיתָה נִפְרָד.

~5

א: גִישַׁת אֵיי בִּי אֶם.

ש: וּמִשְׁפָּט לְסִיוּם?

א: בְּכֹל שְׁאֵלָה, קֹדֶם לְזַהוֹת אֶת הַפֶּרֶק, וְאַחַר כָּךְ אֶת הַמֻשָׂג. וּלְתַרְגֵל בַּאֲתַר סִימוּלַצְיָה שֶׁל אַרְבַּע עֶשְׂרֵה שְׁאֵלוֹת, עַד שֶׁהַזִהוּי נִהְיָה מָהִיר.

ש: בְּהַצְלָחָה בַּמִבְחָן.
