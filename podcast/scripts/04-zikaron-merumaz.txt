# ש = המראיינת (Hila), א = המסביר (Avri). מספרים במילים, בלי מקפים, גרש אסקי.

ש: פֶּרֶק רְבִיעִי: הַזִכָּרוֹן הַמְרֻמָז. מַצֶגֶת אַרְבַּע, הַחֵלֶק הַשֵׁנִי. בִּרְשִׁימַת הַנוֹשְׂאִים לַבְּחִינָה כָּתוּב: זִכָּרוֹן מְרֻמָז לְעֻמַת מְפוֹרָשׁ. אָז נַתְחִיל מֵהַהֶבְדֵל.

א: הַזִכָּרוֹן הַמְפוֹרָשׁ כּוֹלֵל מֵידָע עֻבְדָתִי, שֶׁאֲנַחְנוּ מִשְׁתַמְשִׁים בּוֹ בְּאֹפֶן מוּדָע. לְמָשָׁל: מָה שֵׁם רֹאשׁ הַמֶמְשָׁלָה. הַזִכָּרוֹן הַמְרֻמָז אֵינוֹ מוּדָע. הוּא יָכוֹל לְהִתְבַּסֵס עַל תַסְרִיט פְּעֻלָה אוֹטוֹמָטִי, אוֹ עַל מַשְׁמָעוּיוֹת לֹא מוּדָעוֹת שֶׁל מֻשָׂגִים.

ש: וְאֵיךְ זֶה מִתְחַבֵּר לְכַּהֲנֶמָן?

א: כַּהֲנֶמָן מְדַבֵּר עַל שְׁתֵי מַעֲרָכוֹת, שֶׁמְעֹרָבוֹת בְּכֹל הַחְלָטָה. מַעֲרֶכֶת אַחַת הִיא הַטַיָס הָאוֹטוֹמָטִי: זֶה הַזִכָּרוֹן הַמְרֻמָז. מַעֲרֶכֶת שְׁתַיִם הִיא הַטַיָס הַלֹא אוֹטוֹמָטִי: זֶה הַזִכָּרוֹן הַמְפוֹרָשׁ.

ש: מָה מְאַפְיֵן אֶת הַטַיָס הַלֹא אוֹטוֹמָטִי?

א: הוּא דוֹמֶה לְמָה שֶׁתֵאַרְנוּ בְּמוֹדֵל שְׁלֹשֶׁת הַמַאֲגָרִים. הַצַרְכָן קוֹלֵט חֲמִשָׁה עַד תִשְׁעָה גוּשֵׁי מֵידָע, בְּאֹפֶן מוּדָע וּבִיקוֹרְתִי, וְתוֹךְ מַאֲמָץ מֶנְטָלִי.

ש: וְהַטַיָס הָאוֹטוֹמָטִי?

א: יֵשׁ לוֹ קִבֹּלֶת עֲצוּמָה, וְלָכֵן הוּא מְעַבֵּד כִּמְעַט אֶת כֹּל הָאוֹתוֹת שֶׁבַּסְבִיבָה. הַמֵידָע נִקְלָט בְּצוּרָה אוֹטוֹמָטִית, אָסוֹצְיָאטִיבִית, לֹא מוּדַעַת, וּלְלֹא מַאֲמָץ. וְהַמַסְקָנָה הַחֲשׁוּבָה: אֲפִלוּ גֵרוּי חַלָשׁ, קָלוּשׁ וְהֶקֵפִי יָכוֹל לְהַשְׁפִּיעַ בַּסוֹף עַל הַהַחְלָטָה.

ש: אֵיךְ יוֹדְעִים שֶׁבֶּאֱמֶת יֵשׁ שְׁתֵי מַעֲרָכוֹת?

א: הַסִכּוּם מֵבִיא כַּמָה רְאָיוֹת. הָרִאשׁוֹנָה: אֲנָשִׁים שֶׁלוֹקִים בְּאַמְנֶזְיָה. הֵם שׁוֹכְחִים פְּרָטִים מֵהֶעָבָר, וְלֹא מַצְלִיחִים לִזְכֹּר מֵידָע חָדָשׁ. אֲבָל הֵם כֵּן מְסֻגָלִים לִלְמֹד וְלִזְכֹּר כִּשׁוּרִים מוֹטוֹרִיִים וּתְפִיסָתִיִים.

ש: הַשְׁנִיָה.

א: הַמֶחְקָר שֶׁל פִּיטֶר קֵנִינְג. הוּא בָּדַק מָה קוֹרֶה כְּשֶׁצַרְכָנִים נֶחְשָׂפִים לְמוּתָג חָזָק וּמוֹבִיל, כְּמוֹ קוֹקָה קוֹלָה. בַּמֹחַ שֶׁלָהֶם הִתְעוֹרְרוּ יוֹתֵר הָאֲזוֹרִים שֶׁל הָרֶגֶשׁ, וְהִצְטַמְצְמָה הַפְּעִילוּת בָּאֲזוֹרִים שֶׁל הַחֲשִׁיבָה הָרַצְיוֹנָלִית. וְכֹל זֶה בְּלִי שֶׁהֵם יָדְעוּ.

ש: וּמָה הֵם אָמְרוּ כְּשֶׁשָׁאֲלוּ אוֹתָם?

א: הֵם הִכְחִישׁוּ שֶׁהַמִתוּג הִשְׁפִּיעַ עֲלֵיהֶם. הֵם אָמְרוּ שֶׁהַבְּחִירָה שֶׁלָהֶם מְבֻסֶסֶת רַק עַל תְכוּנוֹת הַמוּצָר. וְעוֹד פְּרָט: הַהַשְׁפָּעָה הַזֹאת נִמְצְאָה בְּעִקָר בַּמוּתָגִים הַחֲזָקִים.

ש: וְיֵשׁ עוֹד שְׁנֵי מֻשָׂגִים שֶׁכְּדַאי לְהַכִּיר.

א: הַתְרָמָה: הָאָדָם מֻכְוָן לְהִתְמַקֵד בְּהֶבֵּטִים מְסֻיָמִים שֶׁל הַגֵרוּי, בְּלִי שֶׁהוּא מוּדָע לְכָךְ. וְאַשְׁלָיַת הָאֱמֶת: אֲנַחְנוּ נוֹטִים לְהַאֲמִין יוֹתֵר לְהַצְהָרוֹת שֶׁכְּבָר שָׁמַעְנוּ.

ש: עַכְשָׁו אַרְבַּע הַמַשְׁמָעוּיוֹת הַשִׁיווּקִיוֹת. הָרִאשׁוֹנָה: מִסְגוּר.

א: הַצַרְכָן מְפָרֵשׁ, בְּאֹפֶן לֹא מוּדָע, הַרְבֵּה גֵרוּיִים שֶׁנִמְצָאִים סְבִיב הַמוּצָר. הֵם חֵלֶק מֵהַמִסְגֶרֶת, וְלֹא מֵהַמוּצָר עַצְמוֹ. הָרֶקַע, הָאֲוִירָה, סִדוּר הַחֲנוּת, הַתְאוּרָה, הָרֵיחַ, הַמוּזִיקָה, הַצְבָעִים, וְגַם הַמִתוּג.

ש: כְּלוֹמַר, גַם מוּתָג הוּא מִסְגֶרֶת.

א: כֵּן. וְהַמִסְגֶרֶת מַשְׁפִּיעָה עַל הַפַּרְשָׁנוּת, וְעַל הָעֵרֶךְ שֶׁהַצַרְכָן נוֹתֵן לַמוּצָר. אֲבָל אִם נִשְׁאַל אֶת הַצַרְכָנִים, הֵם יַכְחִישׁוּ. וְלָכֵן מֶחְקָרִים שֶׁל תִשְׁאוּל עַצְמִי לֹא תוֹפְסִים אֶת הַהַשְׁפָּעָה הַזֹאת.

ש: וְזֹאת הַשְׁאֵלָה עַל הַיַיִן בְּדַף הַתִרְגוּל.

א: אוֹתוֹ יַיִן מֻגָשׁ לִשְׁתֵי קְבוּצוֹת. לְאַחַת בְּכוֹס יֻקְרָתִית, בִּתְאוּרָה נְעִימָה, לְיַד מוּצָרִים יְקָרִים. לִשְׁנִיָה בְּהַגָשָׁה פְּשׁוּטָה. הַקְבוּצָה הָרִאשׁוֹנָה מַעֲרִיכָה אֶת הַיַיִן כְּאֵיכוּתִי יוֹתֵר, וְטוֹעֶנֶת שֶׁהַהַגָשָׁה לֹא הִשְׁפִּיעָה עָלֶיהָ. הַתְשׁוּבָה: הַשְׁפָּעַת מִסְגוּר וְזִכָּרוֹן מְרֻמָז.

ש: וּמָה הָרֶמֶז בִּשְׁאֵלָה כָּזֹאת?

א: שְׁנֵי דְבָרִים יַחַד. הַמוּצָר עַצְמוֹ זֵהֶה, וְרַק הַסְבִיבָה שׁוֹנָה. וְהַצַרְכָנִים מַכְחִישִׁים אֶת הַהַשְׁפָּעָה.

ש: הַמַשְׁמָעוּת הַשְׁנִיָה: צִפִּיוֹת מֻטְמָעוֹת.

א: חֵלֶק מֵהַפַּרְשָׁנוּת שֶׁל הַגֵרוּי בָּא מִמֵידָע שֶׁכְּבָר קַיָם אֵצֶל הַצַרְכָן, וְלֹא בְּהֶכְרֵחַ מוּדָע. הַמֵידָע הַזֶה יוֹצֵר צִפִּיוֹת לְגַבֵּי בִּצוּעֵי הַמוּצָר. זֶה תַהֲלִיךְ שֶׁל טוֹפּ דָאוּן: הַפַּרְשָׁנוּת תְלוּיָה לֹא רַק בַּגֵרוּי, אֶלָא גַם בַּצִפִּיוֹת וּבְיֶדַע קוֹדֵם.

ש: וּמָה הַצִפִּיוֹת עוֹשׂוֹת?

א: שְׁנֵי דְבָרִים. חֲשִׂיפָה סֵלֶקְטִיבִית: אֲנִי שָׂם לֵב רַק לְמֵידָע שֶׁמְחַזֵק עֶמְדָה קוֹדֶמֶת. וּפַרְשָׁנוּת בְּהֶתְאֵם לַצִפִּיוֹת. בְּמֶחְקָר הַפּוּדִינְג, הַצֶבַע הִשְׁפִּיעַ עַל תְפִיסַת הַטַעַם.

ש: הַמַשְׁמָעוּת הַשְׁלִישִׁית: גִישׁוֹת מֻטְבָּעוֹת.

א: לַצַרְכָנִים יֵשׁ עֲמָדוֹת וּתְפִיסוֹת עַל מוּצָרִים וּמוּתָגִים, וְרֻבָּן אֵינָן מוּדָעוֹת. לְמָשָׁל, תַחְלִיף חָלָב אוֹ תַחְלִיף בָּשָׂר יָכוֹל לְהִתָפֵס כְּלֹא טָעִים, בְּאֹפֶן לֹא מוּדָע.

ש: וְאֵיךְ חוֹקְרִים דָבָר כָּזֶה?

א: תִשְׁאוּל יָשִׁיר לֹא מַסְפִּיק. צָרִיךְ מֶחְקְרֵי עֹמֶק: מֶחְקְרֵי הַשְׁלָכָה, תַצְפִּיוֹת, מֶחְקְרֵי נוֹיְרוֹ מַרְקֵטִינְג שֶׁמְנַטְרִים פְּעִילוּת מוֹחִית, וְנִסוּיִים שֶׁל מְדִידָה מְשֻׁתֶפֶת. שָׁם הַצַרְכָנִים לֹא מַסְבִּירִים אֶת הַבְּחִירוֹת שֶׁלָהֶם. הַחוֹקְרִים מַסִיקִים אֶת הַהַשְׁפָּעוֹת מִתוֹךְ הַבְּחִירוֹת עַצְמָן.

ש: וְעֶמְדָה הִיא מֻשָׂג שֶׁהוֹפִיעַ גַם בְּדַף הַתִרְגוּל.

א: כֵּן. צַרְכַן אוֹמֵר: אֲנִי לֹא אוֹהֵב אֶת הַמוּתָג הַזֶה, הוּא נִרְאֶה לִי מְיֻשָׁן, וְלָכֵן כַּנִרְאֶה לֹא אֶבְחַר בּוֹ. זֹאת עֶמְדָה: הַעֲרָכָה אִישִׁית, חִיוּבִית אוֹ שְׁלִילִית, כְּלַפֵּי אוֹבְּיֵקְט, שֶׁמַשְׁפִּיעָה עַל הַנְכוֹנוּת לְהָגִיב כְּלַפָּיו.

ש: הַמַשְׁמָעוּת הָרְבִיעִית: הַנוֹיְרוֹ הִגָיוֹן שֶׁל הַחְלָטוֹת רְכִישָׁה. עֶקְרוֹן הָעֹנֶג וְהַסֵבֶל.

א: כְּשֶׁהַצַרְכָן מְקַבֵּל הַחְלָטָה, קוֹרִים אֶצְלוֹ שְׁנֵי תַהֲלִיכִים בְּמַקְבִּיל, וּבְאֹפֶן לֹא מוּדָע. עֹנֶג, מֵהַצִפִּיָה לַמוּצָר וְלַתוֹעָלוֹת שֶׁלוֹ. וְסֵבֶל, אוֹ כְּאֵב, מֵהָעֲלוּת הַצְפוּיָה.

ש: וְאֵיךְ הוּא מַכְרִיעַ?

א: כְּמוֹ בְּמִשְׁוָאָה. אִם תְפִיסַת הַתוֹעֶלֶת גְבוֹהָה מִתְפִיסַת הָעֲלוּת, הוּא קוֹנֶה. וּכְכֹל שֶׁהָעֵרֶךְ נֵטוֹ, תוֹעֶלֶת פָּחוֹת עֲלוּת, גָבוֹהַ יוֹתֵר, כָּךְ עוֹלֶה הַסִכּוּי לִרְכִישָׁה. לָכֵן הַמְשַׁוֶקֶת מְנַסָה לְהַעֲצִים אֶת הָעֹנֶג, וּלְהַפְחִית אֶת תְפִיסַת הַכְּאֵב.

ש: וְאֶפְשָׁר לְהַפְחִית אֶת תְפִיסַת הָעֲלוּת בְּלִי לְהוֹרִיד אֶת הַמְחִיר?

א: כֵּן. בָּרָמָה הַמְפוֹרֶשֶׁת יֵשׁ אֶת הַמְחִיר הָאֲמִתִי. בָּרָמָה הַמְרֻמֶזֶת אֶפְשָׁר לְשַׁנוֹת אֶת תְפִיסַת הַמְחִיר: אֹפֶן עִצוּב הַמְחִיר, הַמִסְגֶרֶת, וְהַהֶקְשֵׁר. לְמָשָׁל, הַמְחִירִים שֶׁל הַמוּצָרִים שֶׁמֻצָגִים לְיָדוֹ.

ש: וְהַדֻגְמָה הִיא הַסִפְרָה הַשְׂמָאלִית.

א: בַּמֶחְקָר הִצִיגוּ אוֹתָה הֲנָחָה עַל צִנְצֶנֶת רִבָּה, בִּשְׁתֵי דְרָכִים. בִּקְבוּצָה אַחַת: מֵאַרְבָּעָה דוֹלָר וְסֵנְט אֶחָד, לִשְׁלוֹשָׁה דוֹלָר. בַּקְבוּצָה הַשְׁנִיָה: מֵאַרְבָּעָה דוֹלָר, לִשְׁנֵי דוֹלָר וְתִשְׁעִים וְתִשְׁעָה סֵנְט.

ש: וְאֵיפֹה הַהֲנָחָה נִתְפְּסָה כִּגְדוֹלָה יוֹתֵר?

א: בַּקְבוּצָה הַשְׁנִיָה. הַפַּעַר זֵהֶה, אֲבָל שָׁם הַסִפְרָה הַשְׂמָאלִית יָרְדָה מֵאַרְבַּע לִשְׁתַיִם. וְלַסִפְרָה הַשְׂמָאלִית יֵשׁ הַשְׁפָּעָה תְפִיסָתִית חֲזָקָה יוֹתֵר מֵאֲשֶׁר לַסִפְרָה הַיְמָנִית.

ש: וּמָתַי מְשַׁוֵק יִרְצֶה לִפְנוֹת דווקא לַזִכָּרוֹן הַמְרֻמָז?

א: לְפִי הַסִכּוּם: כְּשֶׁהוּא רוֹצֶה לִמְנֹעַ חֲשִׁיבָה בִּיקוֹרְתִית. לְמָשָׁל, לְמַצֵב מוּצָר מָזוֹן כְּטִבְעִי בְּעֶזְרַת סְמָלִים, מִלִים וּצְבָעִים, וְלֹא בְּעֶזְרַת עֻבְדוֹת. אֲבָל גַם כְּשֶׁיֵשׁ עֻבְדוֹת מוּכָחוֹת, אֶפְשָׁר לְחַזֵק אוֹתָן דֶרֶךְ הַזִכָּרוֹן הַמְרֻמָז.

ש: וְיֵשׁ בְּסִכּוּם שְׁאֵלָה עַל רוֹבּוֹט בַּסוּפֶּר פַארְם.

א: הָרֶשֶׁת הִצִיבָה בַּסְנִיף רוֹבּוֹט, שֶׁמְבָרֵךְ אֶת הַלָקוֹחוֹת וְעוֹזֵר לָהֶם לִמְצֹא אֶת הַמַדָף. הַטַעֲנָה: הַמַהֲלָךְ מְקַשֵׁר אֶת הָרֶשֶׁת לְחַדְשָׁנוּת, גַם בְּאֹפֶן מְפוֹרָשׁ וְגַם בְּאֹפֶן מְרֻמָז. הַתְשׁוּבָה: נָכוֹן. בִּמְפוֹרָשׁ, כִּי הַלָקוֹחַ מִשְׁתַמֵשׁ בַּטֶכְנוֹלוֹגְיָה. וּבִמְרֻמָז, כִּי עֶצֶם הַנוֹכְחוּת שֶׁל הָרוֹבּוֹט הִיא אֲמִירָה עֲקִיפָה שֶׁהָרֶשֶׁת חַדְשָׁנִית.

ש: נִשְׁאַר לָנוּ תַסְרִיט פְּעֻלָה.

א: הַזִכָּרוֹן הַסֵמַנְטִי מְקוֹדֵד לְפִי הַמָה. תַסְרִיט פְּעֻלָה מְקוֹדֵד לְפִי הָאֵיךְ: אֵיךְ אֲנַחְנוּ מְבַצְעִים פְּעֻלוֹת. הוּא שׁוֹמֵר מְיֻמָנוּיוֹת, חֲשִׁיבָתִיוֹת וּמוֹטוֹרִיוֹת. הוּא לֹא דוֹרֵשׁ מַאֲמָץ קוֹגְנִיטִיבִי, וְהוּא פּוֹעֵל אוֹטוֹמָטִית, גַם אִם בִּמְקוֹר הַפְּעֻלָה נִלְמְדָה בְּאֹפֶן מוּדָע. כְּמוֹ רְכִיבָה עַל אוֹפַנַיִם, אוֹ מַסְלוּל הֲלִיכָה קָבוּעַ.

ש: וּמִזֶה מַגִיעִים לַדְחִיפָה הַקַלָה.

א: כֵּן. אִם הַצַרְכָנִים פּוֹעֲלִים בְּאֹפֶן אוֹטוֹמָטִי, אֶפְשָׁר לְתַכְנֵן אֶת הַסְבִיבָה כָּךְ שֶׁתְכַוֵן אוֹתָם. הַהַגְדָרָה שֶׁל תֵיילֶר וְסַנְסְטִין: הִתְעָרְבוּת קַלָה, שֶׁמְשַׁנָה אֶת הַהִתְנַהֲגוּת שֶׁל אֲנָשִׁים בְּאֹפֶן צָפוּי, בְּלִי לֶאֱסֹר עֲלֵיהֶם אֶפְשָׁרוּיוֹת, וּבְלִי לְשַׁנוֹת מַשְׁמָעוּתִית אֶת הַתַמְרִיצִים הַכַּלְכָּלִיִים.

ש: דֻגְמָאוֹת מֵהַחֲנוּת.

א: בִּמְדִינוֹת שֶׁבָּהֶן הַהֶגֶה בְּצַד שְׂמֹאל, צַרְכָנִים נוֹטִים לִפְנוֹת יָמִינָה בַּחֲנוּת. לָכֵן מְמַקְמִים בְּצַד יָמִין אֶת הַמוּצָרִים שֶׁרוֹצִים לְקַדֵם. וְעוֹד: צַרְכָנִים לֹא מִתְקַדְמִים עַד הַסוֹף שֶׁל מַעֲבָר אָרֹךְ, אֶלָא אִם יֶשׁ שָׁם מוּצָר עֹגֶן. לָכֵן שָׂמִים מוּצְרֵי עֹגֶן בִּקְצוֹת הַמַעֲבָרִים.

ש: וּמָה הַהֶבְדֵל בֵּין הַכְוָנָה מְפוֹרֶשֶׁת לְהַכְוָנָה מְרֻמֶזֶת?

א: הַכְוָנָה מְפוֹרֶשֶׁת פּוֹנָה לַחֲשִׁיבָה הַמוּדַעַת. לְמָשָׁל אַזְהָרָה עַל קֻפְסַת סִיגַרְיוֹת, אוֹ פֵּרוּט קָלוֹרְיוֹת עַל הָאֲרִיזָה. הַכְוָנָה מְרֻמֶזֶת פּוֹעֶלֶת עַל הַטַיָס הָאוֹטוֹמָטִי.

ש: וְהַדֻגְמָאוֹת לְהַכְוָנָה מְרֻמֶזֶת.

א: מַרְצְפוֹת קְטַנוֹת יוֹתֵר בְּמַסְלוּל הַהֲלִיכָה. נוֹצָר צְלִיל נְקִישָׁה בְּקֶצֶב מָהִיר יוֹתֵר, וְהַצַרְכָן מֵאֵט בְּלִי לָשִׂים לֵב.

א: וּבַר קִנוּחִים שֶׁבּוֹ הָעוּגוֹת חֲתוּכוֹת לִפְרוּסוֹת קְטַנוֹת יוֹתֵר. כָּךְ נִשְׁאַר בַּצַלַחַת מָקוֹם גַם לַפֵּרוֹת, וּצְרִיכַת הַפֵּרוֹת עוֹלָה.

ש: שְׁאֵלוֹת לַדֶרֶךְ. רִאשׁוֹנָה: מִי הוּא הַטַיָס הָאוֹטוֹמָטִי אֵצֶל כַּהֲנֶמָן?

~4

א: מַעֲרֶכֶת אַחַת. הַזִכָּרוֹן הַמְרֻמָז.

ש: שְׁנִיָה: אוֹתוֹ מוּצָר בְּדִיוּק, סְבִיבָה שׁוֹנָה, וְהַצַרְכָנִים מַכְחִישִׁים שֶׁהֻשְׁפְּעוּ.

~4

א: מִסְגוּר, וְזִכָּרוֹן מְרֻמָז.

ש: שְׁלִישִׁית: הַצֶבַע שֶׁל הַפּוּדִינְג מְשַׁנֶה אֶת תְפִיסַת הַטַעַם. אֵיזוֹ מַשְׁמָעוּת זֹאת?

~4

א: צִפִּיוֹת מֻטְמָעוֹת.

ש: רְבִיעִית: לָמָה לֹא מַסְפִּיק לִשְׁאֹל אֶת הַצַרְכָנִים יְשִׁירוֹת מָה הֵם חוֹשְׁבִים עַל הַמוּצָר?

~4

א: כִּי רֹב הָעֲמָדוֹת אֵינָן מוּדָעוֹת. צָרִיךְ מֶחְקְרֵי עֹמֶק, כְּמוֹ מֶחְקְרֵי הַשְׁלָכָה, תַצְפִּיוֹת וְנוֹיְרוֹ מַרְקֵטִינְג.

ש: חֲמִישִׁית: מָהֵם שְׁנֵי הַתַהֲלִיכִים בְּעֶקְרוֹן הָעֹנֶג וְהַסֵבֶל?

~4

א: עֹנֶג מֵהַצִפִּיָה לְתוֹעֶלֶת, וְסֵבֶל מֵהָעֲלוּת הַצְפוּיָה. קוֹנִים כְּשֶׁהַתוֹעֶלֶת הַנִתְפֶּסֶת גְבוֹהָה מֵהָעֲלוּת הַנִתְפֶּסֶת.

ש: שִׁשִׁית: אַזְהָרָה עַל קֻפְסַת סִיגַרְיוֹת. הַכְוָנָה מְפוֹרֶשֶׁת אוֹ מְרֻמֶזֶת?

~4

א: מְפוֹרֶשֶׁת.

ש: וְאַחֲרוֹנָה: מָה הַהַגְדָרָה שֶׁל דְחִיפָה קַלָה?

~4

א: הִתְעָרְבוּת קַלָה שֶׁמְשַׁנָה הִתְנַהֲגוּת בְּאֹפֶן צָפוּי, בְּלִי לֶאֱסֹר אֶפְשָׁרוּיוֹת, וּבְלִי לְשַׁנוֹת מַשְׁמָעוּתִית אֶת הַתַמְרִיצִים הַכַּלְכָּלִיִים.

ש: זֶהוּ לַפֶּרֶק הַזֶה. בַּפֶּרֶק הַבָּא: מְעֹרָבוּת הַצַרְכָנִים, וּמוֹדֵל עֲשָׂהאֵל.
